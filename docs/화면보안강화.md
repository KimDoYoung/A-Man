# A-Man - 화면 보안 및 콘텐츠 유출 방지 강화 방안 (화면보안강화.md)

이 문서는 A-Man 매뉴얼 시스템 내의 노하우 및 업무 흐름 정보가 경쟁사 등으로 유출되는 것을 차단하기 위해, **브라우저 단(Frontend)과 API 통신 단에서 적용할 수 있는 강력한 유출 방지 기법**을 정리한 가이드라인 및 소스코드 수정 계획입니다.

웹 브라우저 환경 특성상 100% 완벽한 유출 차단은 불가능하지만, **"유출에 드는 기술적 비용과 귀찮음을 극대화"**하여 일반 사용자 수준은 물론, IT 기술을 가진 인력의 유출 시도까지 효과적으로 저지하는 것을 목적으로 합니다.

### 2.2 2단계: 개발자 도구(F12) 무력화 및 감시

2. **윈도우 크기 변화 감지 (DevTools 도킹 감지 및 로딩 파싱 중단)**
   * 개발자 도구가 브라우저 하단이나 우측에 열릴 때, 브라우저의 내부 창 크기(`window.innerWidth`, `window.innerHeight`)와 외부 창 크기 격차를 실시간으로 비교하여 F12 사용 여부를 감지합니다.
   * **초기 차단 구체화**:
     * `<head>` 안에서 감지되었을 때는 `document.write`로 보안 경고 화면을 삽입함과 동시에, 뒤이어 파싱될 HTML 본문을 차단하기 위해 **`window.stop()`** 명령으로 브라우저 로딩을 즉시 강제 중단시킵니다.
     * 또한, 경고 화면이 브라우저 최상단에 상주하도록 **`position: fixed; z-index: 9999999`** 스타일을 씌워 본문이 그려지더라도 화면 노출을 원천 가립니다.
     ```javascript
     const threshold = 160;
     function checkDevTools() {
       const widthDev = window.outerWidth - window.innerWidth > threshold;
       const heightDev = window.outerHeight - window.innerHeight > threshold;
       if (widthDev || heightDev) {
         var errorHtml = 
           "<div style='display:flex;flex-direction:column;align-items:center;justify-content:center;height:100vh;font-family:sans-serif;color:#ef4444;background:#fafafa;padding:20px;text-align:center;position:fixed;top:0;left:0;width:100%;height:100%;z-index:9999999;'>" +
           "<h2>⚠️ 보안 경고</h2>" +
           "<p>개발자 도구(F12)가 활성화되어 있어 시스템 접근이 제한됩니다.</p>" +
           "<p>개발자 도구를 완전히 종료하신 후 페이지를 새로고침해 주십시오.</p>" +
           "</div>";
         if (document.body) {
           document.body.innerHTML = errorHtml;
         } else {
           document.write(errorHtml);
           try { window.stop(); } catch(e) {}
           try { document.execCommand("Stop"); } catch(e) {}
         }
         throw new Error("DevTools detected.");
       }
     }
     ```

* **JS 스크립트 보강 (`securityScript` 정의 변경)**:
```java

            "    <script>\n" +
            "        (function() {\n" +
            "            // 1. 우클릭 및 텍스트 선택 시작 방지\n" +
            "            document.addEventListener('contextmenu', function(e) { e.preventDefault(); });\n" +
            "            document.addEventListener('selectstart', function(e) { e.preventDefault(); });\n" +
            "            document.addEventListener('copy', function(e) {\n" +
            "                e.preventDefault();\n" +
            "                alert('보안 정책상 매뉴얼 텍스트를 복사할 수 없습니다.');\n" +
            "            });\n" +
            "            // 2. 단축키 제한 (F12, Ctrl+Shift+I/C/J, Ctrl+S/P/U)\n" +
            "            document.addEventListener('keydown', function(e) {\n" +
            "                if (e.keyCode === 123) { e.preventDefault(); }\n" +
            "                if (e.ctrlKey && e.shiftKey && (e.keyCode === 73 || e.keyCode === 67 || e.keyCode === 74)) { e.preventDefault(); }\n" +
            "                if (e.ctrlKey && (e.keyCode === 83 || e.keyCode === 80 || e.keyCode === 85)) { e.preventDefault(); }\n" +
            "            });\n" +
            "            // 3. F12 개발자 도구 활성화 감시 (도킹창 크기 비교)\n" +
            "            var threshold = 160;\n" +
            "            function checkDevTools() {\n" +
            "                var widthDev = window.outerWidth - window.innerWidth > threshold;\n" +
            "                var heightDev = window.outerHeight - window.innerHeight > threshold;\n" +
            "                if (widthDev || heightDev) {\n" +
            "                    var errorHtml = \n" +
            "                        \"<div style='display:flex;flex-direction:column;align-items:center;justify-content:center;height:100vh;font-family:sans-serif;color:#ef4444;background:#fafafa;padding:20px;text-align:center;position:fixed;top:0;left:0;width:100%;height:100%;z-index:9999999;'>\" +\n" +
            "                        \"<h2>⚠️ 보안 경고</h2>\" +\n" +
            "                        \"<p>개발자 도구(F12)가 활성화되어 있어 시스템 접근이 제한됩니다.</p>\" +\n" +
            "                        \"<p>개발자 도구를 완전히 종료하신 후 페이지를 새로고침해 주십시오.</p>\" +\n" +
            "                        \"</div>\";\n" +
            "                    if (document.body) {\n" +
            "                        document.body.innerHTML = errorHtml;\n" +
            "                    } else {\n" +
            "                        document.write(errorHtml);\n" +
            "                        try { window.stop(); } catch(e) {}\n" +
            "                        try { document.execCommand('Stop'); } catch(e) {}\n" +
            "                    }\n" +
            "                    throw new Error(\"DevTools detected.\");\n" +
            "                }\n" +
            "            }\n" +
            "            // 4. 무한 디버거(Anti-Debugging) 기동\n" +
            "            function startAntiDebug() {\n" +
            "                function debuggerTimer() {\n" +
            "                    try {\n" +
            "                        (function a(i) {\n" +
            "                            if ((\"\" + i / i).length !== 1 || i % 20 === 0) {\n" +
            "                                (function() {}).constructor(\"debugger\")();\n" +
            "                            } else {\n" +
            "                                debugger;\n" +
            "                            }\n" +
            "                            a(++i);\n" +
            "                        })(0);\n" +
            "                    } catch (e) {}\n" +
            "                    setTimeout(debuggerTimer, 200);\n" +
            "                }\n" +
            "                debuggerTimer();\n" +
            "            }\n" +
            "            window.addEventListener('resize', checkDevTools);\n" +
            "            checkDevTools();\n" +
            "            startAntiDebug();\n" +
            "            setInterval(checkDevTools, 500);\n" +
            "        })();\n" +
            "    </script>\n";
```
